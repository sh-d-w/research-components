function getLhsPositionsAndRotations(currentNode) {
  let lhs = currentNode;
  let lhsPositions = new Float32Array(16 * 3);//[];
  let lhsRotations = new Float32Array(16 * 3);//[];
  let lhsScalings = new Float32Array(16 * 3);//[];

  for (let i=0; i < lhsPositions.length; i++) {
    lhsPositions[i] = 0.0;
    lhsRotations[i] = 0.0;
    lhsScalings[i] = 0.0;
  }

  let i = -1;
  while (++i >= 0 && lhs && lhs && lhs.translate) {
    lhsPositions[i * 3 + 0] = lhs.translate.x + 0.0;
    lhsPositions[i * 3 + 1] = lhs.translate.y + 0.0;
    lhsPositions[i * 3 + 2] = lhs.translate.z + 0.0;

    lhsRotations[i * 3 + 0] = lhs.rotate.x + 0.0;
    lhsRotations[i * 3 + 1] = lhs.rotate.y + 0.0;
    lhsRotations[i * 3 + 2] = lhs.rotate.z + 0.0;


    lhsScalings[i * 3 + 0] = lhs.scale.x + 0.0;
    lhsScalings[i * 3 + 1] = lhs.scale.y + 0.0;
    lhsScalings[i * 3 + 2] = lhs.scale.z + 0.0;

    lhs = lhs.lhs;
  }

  return { positions: lhsPositions, rotations: lhsRotations, len: i, scales: lhsScalings };
}

function newItemOrBox(node, name, id, object, scene, color='#FFFFFF') {
  let item;
  let state = "None";
  let material = duplicateShaderMaterial(node, id, color);

  item = BABYLON.Mesh.CreateBox(name + id, id, scene);

  item.material = material;

  return item;// box/item when done.
}

function duplicateShaderMaterial(node, id, color = '#ffffff') {
    let itemColor = hexToRgb(color);

  let material = shaderMaterial.clone(node.name + id);

  material.backFaceCulling = true;
  material.setVector3("myColor", new BABYLON.Vector3(itemColor.r / 255, itemColor.g / 255, itemColor.b / 255));

  let lhs = getLhsPositionsAndRotations(node);
  material.setArray3("uPositions", lhs.positions);
  material.setArray3("uRotations", lhs.rotations);
  material.setArray3("uScales", lhs.scales);
  material.setInt("uActiveCount", lhs.len);

  return material;

}
